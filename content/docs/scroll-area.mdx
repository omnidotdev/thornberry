  ScrollAreaContent,
  ScrollAreaRoot,
  ScrollAreaScrollbar,
  ScrollAreaThumb,

<ScrollArea className="mb-8 h-64 max-w-sm rounded-md border">
  <div className="p-4">
    <h4 className="mb-4 font-medium text-sm">Release notes</h4>
    {Array.from({ length: 30 }, (_, i) => (
      <div key={i} className="border-b py-2 text-sm">
        v1.{30 - i}.0
      </div>
    ))}
  </div>
</ScrollArea>

## Usage

```tsx
import { ScrollArea } from "@/components/ui/scroll-area";
```

```tsx
<ScrollArea className="h-64 rounded-md border">
  <div className="p-4">{/* long content */}</div>
</ScrollArea>
```

Give the scroll area a bounded height (or width, for horizontal scrolling) so its content can overflow.
