---
title: Progress
description: Shows the completion of a task as a linear bar or a circular ring, with determinate and indeterminate states.
---

import {
  CircularProgress,
  Progress,
  ProgressLabel,
  ProgressRange,
  ProgressRoot,
  ProgressTrack,
  ProgressValueText,
} from "@/registry/thornberry/components/progress";

<div className="mb-8 max-w-sm">
  <Progress value={60} label="Uploading" showValueText />
</div>

## Installation

<InstallationTabs filename="progress.json" />

## Usage

```tsx
import { CircularProgress, Progress } from "@/components/ui/progress";
```

```tsx
<Progress value={60} label="Uploading" showValueText />

<CircularProgress value={60} showValueText />
```

Pass `value={null}` to show an indeterminate state when the amount of work is unknown.

#### Custom Range

Values are relative to `min` and `max`, and the value text shows the percentage by default. Pass children to `ProgressValueText` for custom text.

<Tabs items={["Preview", "Code"]} className="mt-4">
  <Tab value="Preview">
    <ProgressRoot value={3} max={5} className="max-w-sm">
      <div className="flex items-center justify-between">
        <ProgressLabel>Steps completed</ProgressLabel>
        <ProgressValueText>3 of 5</ProgressValueText>
      </div>
      <ProgressTrack>
        <ProgressRange />
      </ProgressTrack>
    </ProgressRoot>
  </Tab>
  <Tab value="Code">
