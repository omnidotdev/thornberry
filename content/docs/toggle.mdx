import { Bold, Pin, PinOff, Star } from "lucide-react";

<Toggle className="mb-8" aria-label="Toggle bold">
  <Bold className="size-4" />
</Toggle>

## Installation

<InstallationTabs filename="toggle.json" />

## Usage

```tsx
import { Toggle } from "@/components/ui/toggle";
```

```tsx
<Toggle aria-label="Toggle bold">
  <Bold className="size-4" />
</Toggle>
```

Use `pressed` and `onPressedChange` to control the state, or `defaultPressed` to set the initial state. Give icon-only toggles an `aria-label`.

#### Indicator

`ToggleIndicator` renders its children while on and `fallback` while off.
