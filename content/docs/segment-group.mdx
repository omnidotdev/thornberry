---
title: Segment Group
description: A pill-style segmented control for choosing one option, with an indicator that slides to the selection.
---

import { List, LayoutGrid, CalendarDays } from "lucide-react";
import {
  SegmentGroup,
  SegmentGroupIndicator,
  SegmentGroupItem,
  SegmentGroupItemHiddenInput,
  SegmentGroupItemText,
  SegmentGroupRoot,
} from "@/registry/thornberry/components/segment-group";

<SegmentGroup
  className="mb-8"
  items={["Day", "Week", "Month"]}
  defaultValue="Week"
  aria-label="Calendar range"
/>

## Installation

<InstallationTabs filename="segment-group.json" />

## Usage

```tsx
import { SegmentGroup } from "@/components/ui/segment-group";
```

```tsx
<SegmentGroup items={["Day", "Week", "Month"]} defaultValue="Week" />
```

Items can be strings or `{ value, label, disabled }` objects. Use `value` and `onValueChange` to control the selection.
